      </div>

      title:`รายละเอียดผู้ลงทะเบียน ${esc(id)}`,

    });
    if(res.isDenied) {
      openRegistrationEdit(id);
    } else if(res.dismiss === Swal.DismissReason.cancel) {
      printRegistrationSheet(d);
    }
  }catch(e){err(e)}
}

async function openScoreSummary(wid){
  show('กำลังโหลดข้อมูลคะแนน');
  try {
    const res = await rpc('adminGetWorkScoreSummary', TOKEN, CID, wid);
    hide();
    const {work, authors, assignments, scores, criteria} = res;
    
    const critMap = {};
    criteria.forEach(c => critMap[c.CriteriaID] = c);
    
    const completed = assignments.filter(a => ['COMPLETE', 'LOCKED'].includes(String(a.Status).toUpperCase()));
    
    let html = `<div style="text-align:left; font-size:0.9rem;">
      <div style="margin-bottom:15px;">
        <b>${esc(work.WorkCode || work.WorkID)}</b><br>
        ${esc(work.TitleTH)}<br>
        <small class="muted">${authors.map(x => esc(x.FullName)).join(', ')}</small>
      </div>`;
    
    if (completed.length === 0) {
      html += `<div class="muted" style="text-align:center; padding:20px;">ยังไม่มีกรรมการส่งคะแนน (0/${assignments.length})</div>`;
    } else {
      let totalSum = 0;
      completed.forEach((a, i) => {
        totalSum += Number(a.TotalScore) || 0;
        const myScores = scores.filter(s => s.AssignmentID === a.AssignmentID);
        let scoreDetails = myScores.map(s => {
          const c = critMap[s.CriteriaID] || {};
          return `<div style="display:flex; justify-content:space-between; border-bottom:1px solid #eee; padding:4px 0;">
            <div style="flex:1; padding-right:10px;">${esc(c.CriteriaNameTH || 'เกณฑ์')}</div>
            <div style="font-weight:bold; color:#006D70;">${Number(s.Score)} / ${Number(c.MaxScore)}</div>
          </div>
          ${s.Comment ? `<div style="font-size:0.8rem; color:#666; margin-bottom:8px; padding-left:10px; border-left:2px solid #ccc;">${esc(s.Comment)}</div>` : ''}`;
        }).join('');
        
        html += `<div style="margin-bottom:15px; border:1px solid #ddd; border-radius:6px; padding:10px;">
          <div style="display:flex; justify-content:space-between; margin-bottom:10px;">
            <b>กรรมการท่านที่ ${i+1}</b>
            <span class="badge" style="background:#e3f2fd; color:#0d47a1;">รวม ${Number(a.TotalScore)} คะแนน</span>
          </div>
          ${scoreDetails}
          ${a.RecommendationToAuthor ? `<div style="margin-top:10px; font-size:0.85rem;"><b>ข้อเสนอแนะ:</b><br>${esc(a.RecommendationToAuthor).replace(/\n/g, '<br>')}</div>` : ''}
        </div>`;
      });
      const avg = totalSum / completed.length;
      html += `<div style="text-align:right; font-size:1.1rem; margin-top:15px;">
        <b>คะแนนเฉลี่ย: <span style="color:#006D70;">${avg.toFixed(2)}</span></b>
      </div>`;
    }
    
    html += `</div>`;
    
    Swal.fire({
      title: 'สรุปคะแนนประเมินผลงาน',
      html: html,
      width: 700,
      confirmButtonColor: '#006D70',
      confirmButtonText: 'ปิด'
    });
  } catch(e) {
    err(e);
  }
}

async function uploadLocalFile(wid, regId){
  const file = document.getElementById('localFile').files[0];
  const type = document.getElementById('localType').value;
  if(!file) return Swal.fire('กรุณาเลือกไฟล์','','warning');
  show('กำลังอัปโหลดไฟล์...');
  try {
    const b64 = await new Promise((res,rej)=>{
      const r = new FileReader();
      r.onload = ()=>res(r.result.split(',')[1]);
      r.onerror = rej;
      r.readAsDataURL(file);
    });
    await rpc('adminUploadWorkFiles', TOKEN, CID, wid, regId, {
      method: 'LOCAL',
      fileName: file.name,
      mimeType: file.type || 'application/octet-stream',
      base64: b64,
      fileCategory: type
    });
    hide();
    Swal.fire('อัปโหลดไฟล์สำเร็จ','','success');
    loadWorks();
  } catch(e) { err(e); }
}
