function resolveReviewerCriteria(rawCriteria, work) {
  if (!rawCriteria || !rawCriteria.length) return [];
  const w = work || (CURRENT && CURRENT.work) || {};
  const catCode = String(w.CategoryCode || w.CategoryID || '').trim().toUpperCase();
  const catName = String(w.CategoryName || w.CategoryNameTH || '').trim().toUpperCase();
  const fullWorkCat = (catCode + ' ' + catName).toUpperCase();

  const isResearch = /RESEARCH|วิจัย|R2R/i.test(fullWorkCat);
  const isInnovation = /INNOVAT|นวัตกรรม|สิ่งประดิษฐ์/i.test(fullWorkCat);
  const isCQI = /CQI|BEST\s*PRACTICE/i.test(fullWorkCat);
  const isService = (/SERVICE|บริการ/i.test(fullWorkCat)) && !isCQI;
  const isPrimary = /PRIMARY|ปฐมภูมิ/i.test(fullWorkCat);

  // Group into candidate suites by CategoryID
  const byCat = {};
  rawCriteria.forEach(c => {
    const k = String(c.CategoryID || 'DEFAULT').trim().toUpperCase();
    if (!byCat[k]) byCat[k] = [];
    byCat[k].push(c);
  });

  const suites = [];
  Object.keys(byCat).forEach(k => suites.push(byCat[k]));

  const cleanSuites = [];
  suites.forEach(suite => {
    const seen = new Set();
    let hasDups = false;
    suite.forEach(c => {
      const it = Number(c.ItemNo || c.SortOrder) || 0;
      if (seen.has(it)) hasDups = true;
      seen.add(it);
    });
    if (!hasDups) {
      cleanSuites.push(suite);
    } else {
      const s1 = [], s2 = [];
      const s1Seen = new Set();
      suite.forEach(c => {
        const it = Number(c.ItemNo || c.SortOrder) || 0;
        if (!s1Seen.has(it)) {
          s1.push(c);
          s1Seen.add(it);
        } else {
          s2.push(c);
        }
      });
      if (s1.length) cleanSuites.push(s1);
      if (s2.length) cleanSuites.push(s2);
    }
  });

  let bestSuite = null;
  let bestScore = -99999;

  cleanSuites.forEach(suite => {
    suite.sort((a, b) => (Number(a.ItemNo || a.SortOrder) || 0) - (Number(b.ItemNo || b.SortOrder) || 0));
    const totalMax = suite.reduce((sum, c) => sum + (Number(c.MaxScore) || 0), 0);
    let sScore = 0;
    if (totalMax === 100) sScore += 300;
    else if (totalMax <= 100) sScore += 100;
    else sScore -= 1000;

    suite.forEach(c => {
      const critCat = String(c.CategoryID || '').toUpperCase();
      const title = String(c.CriteriaNameTH || c.CriteriaName || '').toUpperCase();
      const text = critCat + ' ' + title + ' ' + String(c.DescriptionTH || '').toUpperCase();
      if (w.CategoryID && (critCat === String(w.CategoryID).toUpperCase() || critCat.split(/[,;\s]+/).includes(String(w.CategoryID).toUpperCase()))) sScore += 100;
      if (catCode && (critCat === catCode || critCat.split(/[,;\s]+/).includes(catCode))) sScore += 80;

      if (isCQI) {
        if (/CQI|PDCA|ทบทวนสถานการณ์|จุดเริ่มต้นของกิจกรรมการพัฒนาคุณภาพ/i.test(text)) sScore += 60;
        if (critCat === 'CQI' || critCat === 'CAT-4') sScore += 50;
      }
      if (isService) {
        if (/SERVICE|บริการ|ความคุ้มค่า|นวัตกรรมบริการ/i.test(text)) sScore += 60;
        if (critCat === 'SERVICE' || critCat === 'CAT-3') sScore += 50;
        if (/PDCA|จุดเริ่มต้นของกิจกรรมการพัฒนาคุณภาพ/i.test(text)) sScore -= 40;
      }
      if (isPrimary) {
        if (/PRIMARY|ปฐมภูมิ|ชุมชน|เครือข่าย/i.test(text)) sScore += 60;
        if (critCat === 'PRIMARY' || critCat === 'CAT-5') sScore += 50;
      }
      if (isResearch) {
        if (/ระเบียบวิธีวิจัย|จริยธรรมการวิจัย|คำถามวิจัย|RESEARCH|วิจัย/i.test(text)) sScore += 60;
        if (critCat === 'RESEARCH' || critCat === 'CAT-1') sScore += 50;
      }
      if (isInnovation) {
        if (/สิ่งประดิษฐ์|นวัตกรรม|INNOVAT/i.test(text)) sScore += 60;
        if (critCat === 'INNOVATION' || critCat === 'CAT-2') sScore += 50;
      }
      if (!isResearch && !isInnovation && critCat === 'SERVICE_CQI_PRIMARY') sScore += 20;
      if ((isResearch || isInnovation) && critCat === 'RESEARCH_INNOVATION') sScore += 20;
    });

    if (sScore > bestScore) {
      bestScore = sScore;
      bestSuite = suite;
    }
  });

  if (bestSuite && bestSuite.length > 0) {
    return bestSuite;
  }

  // Fallback: pick unique items up to 100 points
  const fallback = [];
  const seenNo = new Set();
  let tot = 0;
  rawCriteria.slice().sort((a, b) => (Number(a.ItemNo || a.SortOrder) || 0) - (Number(b.ItemNo || b.SortOrder) || 0)).forEach(c => {
    const it = Number(c.ItemNo || c.SortOrder) || 0;
    const max = Number(c.MaxScore) || 0;
    if (!seenNo.has(it) && tot + max <= 100) {
      seenNo.add(it);
      fallback.push(c);
      tot += max;
    }
  });
  return fallback.length > 0 ? fallback : rawCriteria;
}

function handleScoreInput(input, maxScore) {
  if (!input) return;
  const max = parseFloat(maxScore) || 0;
  let val = parseFloat(input.value);
  if (!isNaN(val)) {
    if (val > max) {
      input.classList.add('is-invalid');
      input.value = max;
      Swal.fire({
        toast: true,
        position: 'top-end',
        icon: 'warning',
        title: `ห้ามลงคะแนนเกินคะแนนเต็ม (${max} คะแนน)`,
        showConfirmButton: false,
        timer: 2500
      });
    } else if (val < 0) {
      input.value = 0;
      input.classList.remove('is-invalid');
    } else {
      input.classList.remove('is-invalid');
    }
  } else {
    input.classList.remove('is-invalid');
  }
  updateLiveReviewScores();
}

function updateLiveReviewScores() {
  const form = $('#reviewForm');
  if (!form || !CURRENT || !CURRENT.criteria) return;
  const critList = CURRENT.criteria || [];
  let grandTotal = 0;
  let totalMax = 0;
  
  critList.forEach(c => {
    const max = Number(c.MaxScore) || 0;
    totalMax += max;
    const input = form.elements['score_' + c.CriteriaID];
    if (input) {
      let rawVal = input.value.trim();
      let val = parseFloat(rawVal);
      if (!isNaN(val)) {
        if (val < 0) { val = 0; input.value = 0; }
        if (val > max) { val = max; input.value = max; }
        grandTotal += val;
      }
      const valDisplay = !isNaN(val) && rawVal !== '' ? (Math.round(val * 100) / 100) : '-';
      const summaryCrit = $(`#summary_crit_${c.CriteriaID}`);
      if (summaryCrit) summaryCrit.textContent = valDisplay;
    }
  });
  
  if (totalMax > 100) totalMax = 100;
  if (grandTotal > totalMax) grandTotal = totalMax;

  const fmt = n => (Math.round(n * 100) / 100).toFixed(2).replace(/\.00$/, '');
  const totalDisplay = $('#summary_total_score');
  if (totalDisplay) totalDisplay.textContent = fmt(grandTotal);
  const totalMaxDisplay = $('#summary_max_score');
  if (totalMaxDisplay) totalMaxDisplay.textContent = totalMax;
  const bar = $('#summary_score_bar');
  if (bar) {
    const pct = totalMax > 0 ? (grandTotal / totalMax) * 100 : 0;
    bar.style.width = Math.min(100, Math.max(0, pct)) + '%';
  }
}

function toggleOldEvaluationDetails() {
  const panel = $('#oldEvaluationDetails');
  const btn = $('#toggleOldScoresBtn');
  if (!panel || !btn) return;
  const isHidden = panel.style.display === 'none';
  panel.style.display = isHidden ? 'block' : 'none';
  btn.textContent = isHidden ? '🙈 ซ่อนเกณฑ์และคะแนนเดิม' : '👁️ แสดงเกณฑ์และคะแนนเดิม';
}

function copyOldCommentsToForm() {
  if (!CURRENT || !CURRENT.oldEvaluation) return;
  const oe = CURRENT.oldEvaluation;
  const f = $('#reviewForm');
  if (!f) return;
  if (oe.recommendationToAuthor && f.elements['RecommendationToAuthor']) {
    f.elements['RecommendationToAuthor'].value = oe.recommendationToAuthor;
  }
  if (oe.internalComment && f.elements['InternalComment']) {
    f.elements['InternalComment'].value = oe.internalComment;
  }
  if (oe.decision && f.elements['Decision']) {
    f.elements['Decision'].value = oe.decision;
  }
  toast('คัดลอกข้อเสนอแนะเดิมลงในฟอร์มเรียบร้อยแล้ว');
}

function autoFillProportionalScores() {
  if (!CURRENT || !CURRENT.oldEvaluation || !CURRENT.criteria) return;
  const oe = CURRENT.oldEvaluation;
  const oldTotal = Number(oe.totalScore) || 0;
  const oldMax = (oe.criteria || []).reduce((acc, c) => acc + (Number(c.MaxScore) || 0), 0) || 100;
  const ratio = oldMax > 0 ? (oldTotal / oldMax) : 0;
  
  const f = $('#reviewForm');
  if (!f) return;
  
  CURRENT.criteria.forEach(c => {
    const input = f.elements['score_' + c.CriteriaID];
    if (input) {
      const max = Number(c.MaxScore) || 0;
      const propScore = Math.round((max * ratio) * 2) / 2;
      input.value = propScore;
    }
  });
  
  updateLiveReviewScores();
  copyOldCommentsToForm();
  toast('คำนวณคะแนนตามสัดส่วนเดิม (' + Math.round(ratio * 100) + '%) ให้เรียบร้อยแล้ว สามารถปรับแก้ได้ตามต้องการ');
}

async function saveReview(submit) {
  try {
    const f = $('#reviewForm') || document.querySelector('form#reviewForm') || document.forms['reviewForm'];
    if (!f) {
      return Swal.fire({ icon: 'error', title: 'ไม่พบแบบฟอร์มประเมิน', text: 'กรุณาลองรีเฟรชหน้าเว็บหรือเลือกผลงานใหม่อีกครั้ง' });
    }
    if (!CURRENT || !CURRENT.assignment) {
      return Swal.fire({ icon: 'error', title: 'ไม่พบข้อมูลผลงาน', text: 'กรุณากลับไปเลือกผลงานที่ต้องการประเมินใหม่อีกครั้ง' });
    }
    const critList = CURRENT.criteria || [];
    const missing = [];
    const exceeded = [];
    let totalEntered = 0;
    const totalMax = critList.reduce((acc, c) => acc + (Number(c.MaxScore) || 0), 0) || 100;
    
    const scores = critList.map(c => {
      const input = f.elements['score_' + c.CriteriaID] || f.querySelector(`[name="score_${c.CriteriaID}"]`);
      const rawVal = input ? input.value.trim() : '';
      const max = Number(c.MaxScore) || 0;
      if (submit && (rawVal === '' || isNaN(parseFloat(rawVal)))) {
        missing.push(`ข้อ ${c.ItemNo}. ${c.CriteriaNameTH || c.CriteriaName || ''}`);
      }
      const sc = parseFloat(rawVal);
      if (!isNaN(sc) && sc > max) {
        exceeded.push(`ข้อ ${c.ItemNo}. ${c.CriteriaNameTH || c.CriteriaName || ''} (ระบุ ${sc} คะแนน ซึ่งเกินคะแนนเต็ม ${max} คะแนน)`);
      }
      if (!isNaN(sc)) totalEntered += sc;
      const commentInput = f.elements['comment_' + c.CriteriaID] || f.querySelector(`[name="comment_${c.CriteriaID}"]`);
      return {
        CriteriaID: c.CriteriaID,
        Score: rawVal,
        Comment: commentInput ? commentInput.value : ''
      };
    });

    if (exceeded.length > 0) {
      return Swal.fire({
        icon: 'error',
        title: 'คะแนนเกินคะแนนเต็ม!',
        html: `<div style="text-align:left; font-size:0.9rem;">ห้ามลงคะแนนเกินคะแนนเต็มในแต่ละหัวข้อ กรุณาแก้ไขก่อนบันทึก:<br><ul style="margin:8px 0 0 20px; color:#dc2626; font-weight:600;">${exceeded.map(m => `<li>${esc(m)}</li>`).join('')}</ul></div>`
      });
    }
    
    if (submit && missing.length > 0) {
      return Swal.fire({
        icon: 'warning',
        title: 'กรุณากรอกคะแนนให้ครบทุกข้อ',
        html: `<div style="text-align:left; font-size:0.9rem;">พบหัวข้อที่ยังไม่ได้ให้คะแนน:<br><ul style="margin:5px 0 0 20px; color:#b42318">${missing.map(m => `<li>${esc(m)}</li>`).join('')}</ul></div>`
      });
    }
    
    const decisionEl = f.elements['Decision'] || f.querySelector('[name="Decision"]');
    const recEl = f.elements['RecommendationToAuthor'] || f.querySelector('[name="RecommendationToAuthor"]');
    const internalEl = f.elements['InternalComment'] || f.querySelector('[name="InternalComment"]');

    const payload = {
      Scores: scores,
      Decision: decisionEl ? decisionEl.value : '',
      RecommendationToAuthor: recEl ? recEl.value : '',
      InternalComment: internalEl ? internalEl.value : ''
    };
    
    if (submit && !payload.Decision) {
      return Swal.fire({ icon: 'warning', title: 'กรุณาเลือกผลการประเมิน (ผ่าน / แก้ไข / ไม่ผ่าน)' });
    }
    
    if (submit) {
      const x = await Swal.fire({
        icon: 'question',
        title: 'ยืนยันการส่งผลประเมิน?',
        html: `
          <div style="text-align:left; font-size:0.95rem;">
            <p><b>คะแนนรวมสุทธิ: <span style="color:var(--teal); font-size:1.2rem">${totalEntered.toFixed(2).replace(/\.00$/,'')} / ${totalMax} คะแนน</span></b></p>
            <p>ผลการประเมิน: <b>${payload.Decision === 'PASS' ? 'ผ่าน (PASS)' : payload.Decision === 'REVISION' ? 'แก้ไข (REVISION)' : 'ไม่ผ่าน (REJECT)'}</b></p>
            <p style="color:#b42318; font-size:0.85rem;">* หลังส่งผลแล้วข้อมูลจะถูกส่งเข้าสู่ระบบหลักทันที</p>
          </div>
        `,
        showCancelButton: true,
        confirmButtonText: 'ยืนยันส่งผลประเมิน',
        cancelButtonText: 'กลับไปแก้ไข',
        confirmButtonColor: '#006D70'
      });
      if (!x.isConfirmed) return;
    }
    
    show(submit ? 'กำลังส่งผลประเมิน...' : 'กำลังบันทึกร่าง...');
    const aid = CURRENT.assignment.AssignmentID;
    const r = await rpc('reviewerSaveReview', TOKEN, CID, aid, payload, submit);
    
    // Immediately update local cache and state so UI updates in milliseconds
    if (r && r.assignment) {
      if (CURRENT && CURRENT.assignment) {
        Object.assign(CURRENT.assignment, r.assignment);
        if (r.scores) CURRENT.scores = r.scores;
      }
      ASSIGNMENT_CACHE[aid] = CURRENT;
      if (BOOT && BOOT.assignments) {
        const idx = BOOT.assignments.findIndex(x => x.AssignmentID === aid);
        if (idx >= 0) {
          BOOT.assignments[idx] = Object.assign({}, BOOT.assignments[idx], r.assignment);
        }
      }
    }
    
    hide();
    
    Swal.fire({
      icon: 'success',
      title: submit ? 'ส่งผลประเมินเรียบร้อยแล้ว' : 'บันทึกร่างเรียบร้อยแล้ว',
      text: 'คะแนนรวม ' + (r && r.totalScore !== undefined ? r.totalScore : totalEntered.toFixed(2).replace(/\.00$/,'')) + ' / ' + totalMax + ' คะแนน',
      timer: 1800,
      timerProgressBar: true,
      confirmButtonColor: '#006D70',
      confirmButtonText: 'ตกลง'
    });
    
    renderAll();
    
    if (submit) {
      $('#reviewHost').innerHTML = '';
      const grid = $('#assignmentGrid');
      if (grid) grid.scrollIntoView({ behavior: 'smooth' });
    }
    
    // Background sync bootstrap silently without blocking the user
    rpc('reviewerBootstrap', TOKEN, CID).then(freshBoot => {
      if (freshBoot && freshBoot.assignments) {
        BOOT = freshBoot;
        renderAll();
      }
    }).catch(function(){});
  } catch (e) {
    hide();
    console.error('saveReview error:', e);
    Swal.fire({
      icon: 'error',
      title: submit ? 'ส่งผลประเมินไม่สำเร็จ' : 'บันทึกร่างไม่สำเร็จ',
      text: e.message || 'เกิดข้อผิดพลาดในการบันทึกข้อมูล กรุณาลองใหม่อีกครั้ง'
    });
